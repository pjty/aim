    end

        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            dragging = true; upd(i.Position.X)

        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then dragging = false end
    end)
    UserInputService.InputChanged:Connect(function(i)
        if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
            upd(i.Position.X)

local function getPart(plr)
    local ch = plr.Character; if not ch then return nil end
    local hum = ch:FindFirstChildOfClass("Humanoid")
    if not hum or hum.Health <= 0 then return nil end
    local part = ch:FindFirstChild(Settings.TargetPart)
              or ch:FindFirstChild("Head")
              or ch:FindFirstChild("HumanoidRootPart")
              or ch:FindFirstChild("UpperTorso")
              or ch:FindFirstChild("Torso")
    return part
end

local function visible(part)
    if not Settings.WallCheck then return true end
    local p = RaycastParams.new()
    p.FilterType = Enum.RaycastFilterType.Exclude
    p.FilterDescendantsInstances = { LocalPlayer.Character, Camera }
    local r = Workspace:Raycast(Camera.CFrame.Position, part.Position - Camera.CFrame.Position, p)
    if not r then return true end
    return r.Instance:IsDescendantOf(part.Parent)
end

local function getPing()
    local ok, ms = pcall(function()
        return Stats.Network.ServerStatsItem["Data Ping"]:GetValue()
    end)
    if ok and ms then return ms/1000 end
    return 0
end

local function predict(part)
    local pos = part.Position
    if Settings.PingPrediction then
        local v = part.AssemblyLinearVelocity or Vector3.zero
        pos = pos + v * (getPing() * Settings.PredictionStrength * 5)
    end
    if Settings.ProjectilePrediction and Settings.BulletVelocity > 0 then
        local dist = (pos - Camera.CFrame.Position).Magnitude
        local t = dist / Settings.BulletVelocity
        local v = part.AssemblyLinearVelocity or Vector3.zero
        pos = pos + v * t + Vector3.new(0, 0.5 * Settings.BulletGravity * t * t, 0)
    end
    return pos
end

local function passesFilter(plr)
    if plr == LocalPlayer then return false end
    if Settings.TeamCheck and plr.Team == LocalPlayer.Team then return false end
    if Settings.FriendCheck and isFriend(plr) then return false end
    return true
end

local function candidates()
    local out = {}
    local vs = Camera.ViewportSize
    local center = Vector2.new(vs.X/2, vs.Y/2)
    for _,plr in ipairs(Players:GetPlayers()) do
        if passesFilter(plr) then
            local part = getPart(plr)
            if part then
                local sp, on = Camera:WorldToViewportPoint(part.Position)
                if on then
                    local d = (Vector2.new(sp.X, sp.Y) - center).Magnitude
                    if (not Settings.FOVEnabled or d <= Settings.FOV) and visible(part) then
                        table.insert(out, { player=plr, part=part, dist=d })
                    end
                end
            end
        end
    end
    table.sort(out, function(a,b) return a.dist < b.dist end)
    return out
end

local function pick() return candidates()[1] end

local function switchTarget()
    local l = candidates(); if #l == 0 then return end
    local idx = 1
    if State.LockedTarget then
        for i,c in ipairs(l) do
            if c.player == State.LockedTarget.player then idx = i+1; break end
        end
    end
    if idx > #l then idx = 1 end
    State.LockedTarget = l[idx]
end

UserInputService.InputBegan:Connect(function(input, gpe)
    if State.Rebinding or gpe then return end
    if Settings.AimKey == input.UserInputType or Settings.AimKey == input.KeyCode then
        State.Aiming = true
        if Settings.LockMode then
            local c = pick(); if c then State.LockedTarget = c end
        end
    end
    if input.KeyCode == Settings.SwitchKey and Settings.LockMode then switchTarget() end
end)
UserInputService.InputEnded:Connect(function(input)
    if Settings.AimKey == input.UserInputType or Settings.AimKey == input.KeyCode then
        State.Aiming = false; State.LockedTarget = nil
    end
end)

-- AIM LOOP

RunService.RenderStepped:Connect(function()
    if not Settings.Enabled or not State.Aiming then return end
    local target
    if Settings.LockMode then
        if State.LockedTarget then
            local p = getPart(State.LockedTarget.player)
            if p and visible(p) then target = { part=p } else State.LockedTarget = nil end
        end
        if not target then local c = pick(); if c then State.LockedTarget = c; target = c end end
    else
        target = pick()
    end
    if not target then return end
    local aimPos = predict(target.part)
    local cur = Camera.CFrame
    local goal = CFrame.new(cur.Position, aimPos)
    if Settings.HardLock then
        Camera.CFrame = goal
    else
        Camera.CFrame = cur:Lerp(goal, math.clamp(1 - Settings.Smoothness, 0.01, 1))
    end
end)

local function silentTargetPos()
    if not Settings.SilentAim then return nil end
    local c = pick(); if not c then return nil end
    return predict(c.part), c.part
end

local ok = pcall(function()
    if not (hookmetamethod and getnamecallmethod and checkcaller) then error("no executor") end
    local oldNamecall
    oldNamecall = hookmetamethod(game, "__namecall", function(self, ...)
        local method = getnamecallmethod()
        if not checkcaller() and Settings.SilentAim then
            local args = { ... }
            if self == Workspace and (method == "Raycast" or method == "FindPartOnRay"
                or method == "FindPartOnRayWithIgnoreList" or method == "FindPartOnRayWithWhitelist") then
                local origin = args[1]
                if typeof(origin) == "CFrame" then origin = origin.Position end
                if typeof(origin) == "Vector3" then
                    local pos = silentTargetPos()
                    if pos then
                        if method == "Raycast" then args[2] = (pos - origin)
                        else args[1] = Ray.new(origin, (pos - origin)) end
                        return oldNamecall(self, table.unpack(args))
                    end
                end
            end
        end
        return oldNamecall(self, ...)
    end)
    local oldIndex
    oldIndex = hookmetamethod(game, "__index", function(self, key)
        if not checkcaller() and Settings.SilentAim and typeof(self)=="Instance" and self:IsA("Mouse") then
            if key == "Hit" then
                local pos = silentTargetPos(); if pos then return CFrame.new(pos) end
            elseif key == "Target" then
                local _, part = silentTargetPos(); if part then return part end
            end
        end
        return oldIndex(self, key)
    end)
end)

local ESPCache = {} -- [player] = { gui elements... }

local SKELETON_LINKS = {
    {"Head","UpperTorso"},{"UpperTorso","LowerTorso"},
    {"UpperTorso","LeftUpperArm"},{"LeftUpperArm","LeftLowerArm"},{"LeftLowerArm","LeftHand"},
    {"UpperTorso","RightUpperArm"},{"RightUpperArm","RightLowerArm"},{"RightLowerArm","RightHand"},
    {"LowerTorso","LeftUpperLeg"},{"LeftUpperLeg","LeftLowerLeg"},{"LeftLowerLeg","LeftFoot"},
    {"LowerTorso","RightUpperLeg"},{"RightUpperLeg","RightLowerLeg"},{"RightLowerLeg","RightFoot"},
}

local function newLine()
    local f = Instance.new("Frame")
    f.BorderSizePixel = 0
    f.BackgroundColor3 = Color3.fromRGB(255,255,255)
    f.AnchorPoint = Vector2.new(0,0.5)
    f.Parent = ESPGui
    return f
end
local function drawLine(f, a, b, color)
    local dx, dy = b.X - a.X, b.Y - a.Y
    local len = math.sqrt(dx*dx + dy*dy)
    f.Size = UDim2.fromOffset(len, 1)
    f.Position = UDim2.fromOffset(a.X, a.Y)
    f.Rotation = math.deg(math.atan2(dy, dx))
    f.BackgroundColor3 = color
    f.Visible = true
end

local function ensureESP(plr)
    if ESPCache[plr] then return ESPCache[plr] end
    local box = Instance.new("Frame")
    box.BackgroundTransparency = 1
    box.BorderSizePixel = 0
    box.Parent = ESPGui
    local stroke = Instance.new("UIStroke", box)
    stroke.Thickness = 1
    stroke.Color = Settings.ColorEnemy

    local name = Instance.new("TextLabel")
    name.BackgroundTransparency = 1
    name.Font = Enum.Font.GothamSemibold
    name.TextSize = 12
    name.TextStrokeTransparency = 0.4
    name.TextColor3 = Color3.fromRGB(240,240,240)
    name.Parent = ESPGui

    local dist = name:Clone(); dist.Parent = ESPGui; dist.TextSize = 11

    local hpBG = Instance.new("Frame")
    hpBG.BackgroundColor3 = Color3.fromRGB(0,0,0)
    hpBG.BorderSizePixel = 0
    hpBG.Parent = ESPGui
    local hp = Instance.new("Frame")
    hp.BackgroundColor3 = Color3.fromRGB(80,220,100)
    hp.BorderSizePixel = 0
    hp.Parent = hpBG

    local tracer = newLine()

    local cham = Instance.new("Highlight")
    cham.FillColor = Settings.ColorEnemy
    cham.OutlineColor = Color3.new(1,1,1)
    cham.FillTransparency = 0.6
    cham.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
    cham.Parent = ESPGui

    local skel = {}
    for i=1,#SKELETON_LINKS do skel[i] = newLine() end

    ESPCache[plr] = { box=box, stroke=stroke, name=name, dist=dist,
        hpBG=hpBG, hp=hp, tracer=tracer, cham=cham, skel=skel }
    return ESPCache[plr]
end

local function clearESP(plr)
    local e = ESPCache[plr]; if not e then return end
    for _,v in pairs(e) do
        if typeof(v) == "table" then
            for _,f in ipairs(v) do f:Destroy() end
        elseif typeof(v) == "Instance" then
            v:Destroy()
        end
    end
    ESPCache[plr] = nil
end

local function hideESP(e)
    e.box.Visible = false; e.name.Visible = false; e.dist.Visible = false
    e.hpBG.Visible = false; e.tracer.Visible = false
    e.cham.Adornee = nil
    for _,l in ipairs(e.skel) do l.Visible = false end
end

Players.PlayerRemoving:Connect(clearESP)

RunService.RenderStepped:Connect(function()
    for _,plr in ipairs(Players:GetPlayers()) do
        if plr == LocalPlayer then continue end
        if not Settings.ESP_Enabled then
            if ESPCache[plr] then hideESP(ESPCache[plr]) end
            continue
        end
        if Settings.ESP_TeamCheck and plr.Team == LocalPlayer.Team then
            if ESPCache[plr] then hideESP(ESPCache[plr]) end
            continue
        end
        local ch = plr.Character
        local hum = ch and ch:FindFirstChildOfClass("Humanoid")
        local root = ch and (ch:FindFirstChild("HumanoidRootPart") or ch:FindFirstChild("Torso"))
        local head = ch and ch:FindFirstChild("Head")
        if not (ch and hum and root and hum.Health > 0) then
            if ESPCache[plr] then hideESP(ESPCache[plr]) end
            continue
        end
        local e = ensureESP(plr)
        local color = Settings.ColorEnemy
        if State.LockedTarget and State.LockedTarget.player == plr then color = Settings.ColorLocked end
        e.stroke.Color = color

        local topPos, topOn = Camera:WorldToViewportPoint((head or root).Position + Vector3.new(0,1,0))
        local botPos = Camera:WorldToViewportPoint(root.Position - Vector3.new(0,3,0))
        local h = math.abs(botPos.Y - topPos.Y)
        local w = h * 0.55

        -- Box
        if Settings.ESP_Box and topOn then
            e.box.Visible = true
            e.box.Position = UDim2.fromOffset(topPos.X - w/2, topPos.Y)
            e.box.Size = UDim2.fromOffset(w, h)
        else e.box.Visible = false end

        -- Name
        if Settings.ESP_Name and topOn then
            e.name.Visible = true; e.name.Text = plr.Name
            e.name.Position = UDim2.fromOffset(topPos.X - 60, topPos.Y - 16)
            e.name.Size = UDim2.fromOffset(120, 14)
            e.name.TextColor3 = color
        else e.name.Visible = false end

        -- Distance
        if Settings.ESP_Distance and topOn and LocalPlayer.Character then
            local hrp = LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
            if hrp then
                local d = math.floor((hrp.Position - root.Position).Magnitude)
                e.dist.Visible = true; e.dist.Text = d.." studs"
                e.dist.Position = UDim2.fromOffset(topPos.X - 60, botPos.Y + 2)
                e.dist.Size = UDim2.fromOffset(120, 14)
                e.dist.TextColor3 = color
            end
        else e.dist.Visible = false end

        -- Health
        if Settings.ESP_Health and topOn then
            local pct = math.clamp(hum.Health / hum.MaxHealth, 0, 1)
            e.hpBG.Visible = true
            e.hpBG.Position = UDim2.fromOffset(topPos.X - w/2 - 6, topPos.Y)
            e.hpBG.Size = UDim2.fromOffset(3, h)
            e.hp.Position = UDim2.fromScale(0, 1 - pct)
            e.hp.Size = UDim2.fromScale(1, pct)
            e.hp.BackgroundColor3 = Color3.fromRGB(math.floor(255*(1-pct)), math.floor(220*pct), 80)
        else e.hpBG.Visible = false end

        -- Tracer
        if Settings.ESP_Tracer and topOn then
            local vs = Camera.ViewportSize
            drawLine(e.tracer, Vector2.new(vs.X/2, vs.Y), Vector2.new(botPos.X, botPos.Y), color)
        else e.tracer.Visible = false end

        -- Chams
        if Settings.ESP_Chams then
            e.cham.Adornee = ch; e.cham.FillColor = color
        else e.cham.Adornee = nil end

        -- Skeleton
        if Settings.ESP_Skeleton then
            for i,link in ipairs(SKELETON_LINKS) do
                local a = ch:FindFirstChild(link[1]); local b = ch:FindFirstChild(link[2])
                if a and b then
                    local ap, ao = Camera:WorldToViewportPoint(a.Position)
                    local bp, bo = Camera:WorldToViewportPoint(b.Position)
                    if ao and bo then
                        drawLine(e.skel[i], Vector2.new(ap.X, ap.Y), Vector2.new(bp.X, bp.Y), color)
                    else e.skel[i].Visible = false end
                else e.skel[i].Visible = false end
            end
        else
            for _,l in ipairs(e.skel) do l.Visible = false end
        end
    end
end)
